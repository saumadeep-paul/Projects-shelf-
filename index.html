Arduino Robot

arduino_robot.ino

Bluetooth-controlled four-legged robot using five servo motors.

// Arduino Robot
#include "Servo.h"
#include "SoftwareSerial.h"

// Define servo objects for the 4 legs and lift
Servo frontLeft;
Servo frontRight;
Servo rearLeft;
Servo rearRight;
Servo motorup;

// Define Arduino pin connections
const int PIN_FL = 3;
const int PIN_FR = 5;
const int PIN_RL = 6;
const int PIN_RR = 9;
const int lift = 11;

// Movement angle configuration
const int REST_ANGLE = 90;    // Neutral middle position
const int SWING_FORWARD = 60; // Forward stroke angle
const int SWING_BACK = 120;  // Backward stroke angle
const int robot_lift = 0;
const int down = 75;

SoftwareSerial mySerial(10, 12); // (RX, TX)

// Variable to persist the active state across main loop iterations
char currentCommand = 'S';

void setup()
{
  Serial.begin(9600);
  mySerial.begin(9600);

  // Attach servos to designated digital pins
  frontLeft.attach(PIN_FL);
  frontRight.attach(PIN_FR);
  rearLeft.attach(PIN_RL);
  rearRight.attach(PIN_RR);
  motorup.attach(lift);

  // Set all legs to neutral starting position
  frontLeft.write(REST_ANGLE);
  frontRight.write(REST_ANGLE);
  rearLeft.write(REST_ANGLE);
  rearRight.write(REST_ANGLE);
  motorup.write(down);

  delay(1000); // Wait 1 second before starting movement
}

void loop()
{
  // 1. Check if new serial data has arrived
  if (mySerial.available() > 0)
  {
    char val = mySerial.read();

    // Only update state for valid control commands (filters out '\r' or '\n')
    if (val == 'F' || val == 'B' || val == 'S' || val == 'R' || val == 'L' )
    {
      currentCommand = val;
      Serial.print("Command changed to: ");
      Serial.println(currentCommand);
    }
  }

  // 2. Execute continuous movement based on the active command state
  if (currentCommand == 'F')
  {
    // Forward Step
    frontLeft.write(SWING_FORWARD);
    rearRight.write(SWING_FORWARD);
    frontRight.write(SWING_BACK);
    rearLeft.write(SWING_BACK);
    delay(400);

    motorup.write(robot_lift);
    delay(400);

    frontLeft.write(SWING_BACK);
    rearRight.write(SWING_BACK);
    frontRight.write(SWING_FORWARD);
    rearLeft.write(SWING_FORWARD);
    delay(400);

    motorup.write(down);
    delay(400);
  }
  else if (currentCommand == 'B')
  {
    // Backward Step
    frontLeft.write(SWING_BACK);
    rearRight.write(SWING_BACK);
    frontRight.write(SWING_FORWARD);
    rearLeft.write(SWING_FORWARD);
    delay(400);

    motorup.write(robot_lift);
    delay(400);

    (SWING_BACK);
    rearRight.write(SWING_FORWARD);
    frontLeft.write(SWING_FORWARD);
    rearLeft.write(SWING_BACK);
    frontRight.write(SWING_BACK);
    delay(400);

    motorup.write(down);
    delay(400);
  }
  else if (currentCommand == 'S')
  {
    // Stop position
    frontLeft.write(REST_ANGLE);
    rearRight.write(REST_ANGLE);
    frontRight.write(REST_ANGLE);
    rearLeft.write(REST_ANGLE);
    motorup.write(down);
  }
    else if (currentCommand == 'R')
  {
    // Forward Step
    frontLeft.write(SWING_BACK);
    rearRight.write(SWING_BACK);
    frontRight.write(SWING_BACK);
    rearLeft.write(SWING_BACK);
    delay(400);

    motorup.write(robot_lift);
    delay(400);

    frontLeft.write(SWING_FORWARD);
    rearRight.write(SWING_FORWARD);
    frontRight.write(SWING_FORWARD);
    rearLeft.write(SWING_FORWARD);
    delay(400);

    motorup.write(down);
    delay(400);
    
}else if (currentCommand == 'L')

    // Forward Step

ESP32 Air Mouse

esp32_air_mouse.ino

ESP32 air mouse using an MPU6050 motion sensor.

// ESP32 Air Mouse
#include "Wire.h"
#include "HijelHID_BLEMouse.h"

// Define button pins (matching D25 and D26)
#define LEFT_BTN_PIN 25
#define RIGHT_BTN_PIN 26

// Initialize Bluetooth Mouse
HijelBLEMouse mouse("ESP32 Air Mouse", "Custom");

// MPU6500 I2C Address (0x68 if AD0 is grounded, 0x69 if pulled high)
const uint8_t MPU_ADDR = 0x68; 

// Tuning variables for mouse movement
const float deadzone = 0.05;     // Ignore minor jitters
const int sensitivity = 15;      // Cursor speed multiplier

// Variables to track button states
bool leftWasPressed = false;
bool rightWasPressed = false;

void setup() {
  Serial.begin(115200);

  // Start I2C communication on GPIO 21 (SDA) and GPIO 22 (SCL)
  Wire.begin(21, 22);

  // Start Bluetooth Mouse first so it broadcasts immediately
  mouse.begin();
  Serial.println("Bluetooth started. Looking for device...");

  // Setup buttons with internal pull-up resistors
  pinMode(LEFT_BTN_PIN, INPUT_PULLUP);
  pinMode(RIGHT_BTN_PIN, INPUT_PULLUP);

  // --- Initialize MPU6500 via Raw I2C ---
  // 1. Wake up MPU6500 (Clear sleep mode bit in PWR_MGMT_1 register)
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B); // PWR_MGMT_1 register
  Wire.write(0x00); // Set to zero to wake up
  Wire.endTransmission();

  // 2. Set Gyroscope range to ±500 deg/s (GYRO_CONFIG register 0x1B)
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x1B);
  Wire.write(0x08); // 0x08 sets range to 500 deg/s
  Wire.endTransmission();

  Serial.println("MPU6500 Initialized Successfully!");
}

void loop() {
  // Only process data if the phone/PC is actively connected
  if (mouse.isConnected()) {
    
    /* -----------------------------------
       1. Handle Mouse Movement (Gyroscope)
    ----------------------------------- */
    // Request raw gyroscope data starting from register 0x43
    Wire.beginTransmission(MPU_ADDR);
    Wire.write(0x43); 
    Wire.endTransmission(false);
    Wire.requestFrom(MPU_ADDR, (uint8_t)6); // Request 6 bytes (X_high, X_low, Y_high, Y_low, Z_high, Z_low)
// remove // before 8
    if (Wire.available() >= 6) {
      int16_t gyroX_raw = (Wire.read() < deadzone || gyroZ < -deadzone) {
        vx = -gyroZ * sensitivity; 
      }

      if (gyroX > deadzone || gyroX < -deadzone) {
        vy = -gyroX * sensitivity; 
      }

      if (vx != 0 || vy != 0) {
        mouse.move(vx, vy);
        
        // Debug output to Serial Monitor
        Serial.print("Moving X: "); Serial.print(vx);
        Serial.print(" | Moving Y: "); Serial.println(vy);
      }
    }

    /* -----------------------------------
       2. Handle Button Clicks
    ----------------------------------- */
    bool leftIsPressed = (digitalRead(LEFT_BTN_PIN) == LOW);
    bool rightIsPressed = (digitalRead(RIGHT_BTN_PIN) == LOW);

    // Left Click logic
    if (leftIsPressed && !leftWasPressed) {
      mouse.press(MouseButton::Left);
      leftWasPressed = true;
      Serial.println("LEFT Button Pressed");
    } else if (!leftIsPressed && leftWasPressed) {
      mouse.release(MouseButton::Left);
      leftWasPressed = false;
    }

    // Right Click logic
    if (rightIsPressed && !rightWasPressed) {
      mouse.press(MouseButton::Right);
      rightWasPressed = true;
      Serial.println("RIGHT Button Pressed");
    } else if (!rightIsPressed && rightWasPressed) {
      mouse.release(MouseButton::Right);
      rightWasPressed = false;
    }

    delay(15); // Small stabilization delay
  }
}
